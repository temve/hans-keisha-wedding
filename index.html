Dual Upload (Framed + Without Frame)

Automatically saves both Framed & Clean copies to separate Drive folders.

Google Apps Script (Dual-Folder Auto-Sorter) 
function doPost(e) {
  var ROOT_FOLDER_ID = "PASTE_YOUR_GOOGLE_DRIVE_FOLDER_ID_HERE";
  var REQUIRED_KEY = "";

  var data = JSON.parse(e.postData.contents);
  if (REQUIRED_KEY && data.accessKey !== REQUIRED_KEY) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "unauthorized" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var rootFolder = DriveApp.getFolderById(ROOT_FOLDER_ID);
  var subName = (data.folderType === "Clean" || data.variant === "clean")
    ? "Original (No Frame)"
    : "Framed";
  var folders = rootFolder.getFoldersByName(subName);
  var targetFolder = folders.hasNext() ? folders.next() : rootFolder.createFolder(subName);

  var bytes = Utilities.base64Decode(data.base64Data);
  var blob = Utilities.newBlob(bytes, data.mimeType, data.filename);
  var file = targetFolder.createFile(blob);

  return ContentService
    .createTextOutput(JSON.stringify({ status: "success", fileId: file.getId(), name: file.getName() }))
    .setMimeType(ContentService.MimeType.JSON);
}
